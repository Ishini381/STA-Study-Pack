# Create zoology-related dataset
fish_data <- data.frame(
  Species = c("Salmon", "Trout", "Carp", "Pike", "Perch", "Bass", "Catfish", "Eel", "Herring", "Mackerel"),
  Body_Length_cm = c(75, 68, 55, 90, 45, 50, 60, 100, 25, 35),
  Body_Weight_g = c(4500, 3200, 2500, 5000, 1200, 1500, 2200, 3500, 300, 500),
  Fin_Length_cm = c(20, 18, 15, 22, 12, 14, 13, 25, 8, 10)
)

# View dataset
fish_data

# Remove species column
fish_numeric <- fish_data[, -1]

# Scale data
fish_scaled <- scale(fish_numeric)

head(fish_scaled)

###########

# Calculate distance matrix
dist_matrix <- dist(fish_scaled, method = "euclidean")

# Perform hierarchical clustering
hc <- hclust(dist_matrix, method = "ward.D2")

# Plot dendrogram
plot(hc, labels = fish_data$Species, main = "Hierarchical Clustering Dendrogram", xlab = "", sub = "")
rect.hclust(hc, k = 3, border = "red")  # Draw cluster boundaries

###########

set.seed(123)  # For reproducibility
kmeans_result <- kmeans(fish_scaled, centers = 3, nstart = 25)

# Cluster assignments
fish_data$Cluster <- as.factor(kmeans_result$cluster)

fish_data

###########
library(ggplot2)

ggplot(fish_data, aes(x = Body_Length_cm, y = Body_Weight_g, color = Cluster, 
                      label = Species)) +
  geom_point(size = 3) +
  geom_text(vjust = -0.5) +
  labs(title = "K-means Clustering of Fish", x = "Body Length (cm)", y = "Body Weight (g)") +
  theme_minimal()

###########
# comparing methods

table(Hierarchical = cutree(hc, k = 3), KMeans = kmeans_result$cluster)
